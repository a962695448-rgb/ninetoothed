"""Sample the current test's Python stack without asynchronous signal dumps."""

import hashlib
import json
import os
import sys
import threading
import time
import traceback
from pathlib import Path

import pytest


@pytest.hookimpl(hookwrapper=True)
def pytest_runtest_call(item):
    directory = Path(os.environ["NINE_STACK_DIR"])
    path = directory / (
        "stack-" + hashlib.sha256(item.nodeid.encode()).hexdigest()[:12] + ".jsonl"
    )
    owner = threading.get_ident()
    stopped = threading.Event()
    started = time.monotonic()

    def sample():
        while not stopped.wait(60):
            frame = sys._current_frames().get(owner)
            record = {
                "test": item.nodeid,
                "elapsed_seconds": time.monotonic() - started,
                "stack": traceback.format_stack(frame) if frame is not None else [],
            }
            with path.open("a") as output:
                output.write(json.dumps(record) + "\n")

    thread = threading.Thread(target=sample, daemon=True)
    thread.start()
    try:
        yield
    finally:
        stopped.set()
        thread.join(timeout=2)
